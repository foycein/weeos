cmake_minimum_required(VERSION 3.20)
project(WeeOS LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_compile_options(/W4 /permissive- /utf-8 /EHsc)
include_directories(source)

add_subdirectory(source/Formats)
add_subdirectory(source/Media)
add_subdirectory(source/Boot)
add_subdirectory(source/Input)
add_subdirectory(source/Ios)
add_subdirectory(source/Platform)
add_subdirectory(source/Graphics)
add_subdirectory(source/Effects)
add_subdirectory(source/Text)
add_subdirectory(source/SharpArt)
add_subdirectory(source/Layout)
add_subdirectory(source/Mii)
add_subdirectory(source/Settings)
add_subdirectory(source/SystemMenu)
add_subdirectory(source/Browser)
add_subdirectory(source/Channels)
add_subdirectory(source/WeeOS)
add_subdirectory(source/Sound)
add_subdirectory(tools/NandTool)
add_subdirectory(tools/Setup)
add_subdirectory(tools/ControllerCheck)

# The tests and the tools of development are only in the development repository.
if(EXISTS "${CMAKE_SOURCE_DIR}/tests")
    include_directories(tests)
    enable_testing()
    add_subdirectory(tests)
endif()

set(DevelopmentParts tools/PrepareWiiFiles tools/SoundCheck tools/TextureDump tools/GraphicsDemo)
foreach(developmentPart IN LISTS DevelopmentParts)
    if(EXISTS "${CMAKE_SOURCE_DIR}/${developmentPart}")
        add_subdirectory(${developmentPart})
    endif()
endforeach()
